package jar;

import java.time.Duration;

import org.openqa.selenium.Alert;
import org.openqa.selenium.By;
import org.openqa.selenium.JavascriptExecutor;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;

public class test {

    // Method to pause execution for 5 seconds
    public static void waitTime() throws InterruptedException {
        Thread.sleep(5000);
    }

    public static void main(String[] args) throws InterruptedException {

        // Set ChromeDriver path
        System.setProperty("webdriver.chrome.driver",
                "E:\\Downloads\\chromedriver-win64\\chromedriver-win64\\chromedriver.exe");

        // Launch Chrome browser
        WebDriver driver = new ChromeDriver();

        // Open student management website
        driver.get("http://localhost/student");

        // Maximize browser window
        driver.manage().window().maximize();

        // Set page load timeout
        driver.manage().timeouts().pageLoadTimeout(Duration.ofSeconds(10));

        waitTime();

        // Open admin login page
        driver.get("http://localhost/student/admin");

        waitTime();

        // Login credentials
        String username = "admin";
        String password = "admin@123";

        // Validate login fields before entering data
        if (username.isEmpty()) {
            System.out.println("Username is empty");

        } else if (password.isEmpty()) {
            System.out.println("Password is empty");

        } else {

            // Enter username
            driver.findElement(By.name("username")).sendKeys(username);
            waitTime();

            // Enter password
            driver.findElement(By.name("password")).sendKeys(password);
            waitTime();

            // Click login button
            driver.findElement(By.xpath("//button[@type='submit']")).click();

            System.out.println("Login Success");
            waitTime();
        }

        // Navigate to Students page
        driver.findElement(By.linkText("Students")).click();
        waitTime();

        // Validate current URL
        String expectedURL = "http://localhost/student/admin/student.php";
        String currentURL = driver.getCurrentUrl();

        if (currentURL.equals(expectedURL)) {
            System.out.println("URL Validation Success");
        } else {
            System.out.println("URL Validation Fail");
        }

        waitTime();

        // Search for a student
        WebElement search = driver.findElement(By.xpath("//input[@type='search']"));
        search.sendKeys("Ad");
        waitTime();

        // Click New Student button
        driver.findElement(By.id("new_std_btn")).click();
        waitTime();

        // Enter first name
        driver.findElement(By.name("firstname")).sendKeys("Stanley");
        waitTime();

        // Enter last name
        driver.findElement(By.name("lastname")).sendKeys("Lambert");
        waitTime();

        // Enter grade
        driver.findElement(By.name("grade")).sendKeys("Grade 11");
        waitTime();

        // Enter phone number
        driver.findElement(By.name("phone")).sendKeys("0777889090");
        waitTime();

        // Save new student
        driver.findElement(By.name("save")).click();
        waitTime();

        // Handle save confirmation alert
        Alert addAlert = driver.switchTo().alert();

        System.out.println("Student Saving Alert Message : " + addAlert.getText());

        waitTime();

        // Click OK on alert
        addAlert.accept();
        waitTime();

        // ==========================
        // DELETE STUDENT RECORD
        // ==========================

        // Locate first delete button
        WebElement deleteBtn = driver.findElement(
                By.xpath("(//button[contains(@class,'remove_student')])[1]"));

        // Click delete button using JavaScript
        ((JavascriptExecutor) driver)
                .executeScript("arguments[0].click();", deleteBtn);

        waitTime();

        // Handle delete confirmation alert
        Alert deleteAlert = driver.switchTo().alert();

        System.out.println("Student Deleting Alert Message : "
                + deleteAlert.getText());

        waitTime();

        // Cancel deletion
        deleteAlert.dismiss();
        waitTime();

        // ==========================
        // UPDATE STUDENT RECORD
        // ==========================

        // Locate first edit button
        WebElement editBtn = driver.findElement(
                By.xpath("(//button[contains(@class,'edit_student')])[1]"));

        // Click edit button using JavaScript
        ((JavascriptExecutor) driver)
                .executeScript("arguments[0].click();", editBtn);

        waitTime();

        // Clear existing last name
        WebElement lname = driver.findElement(By.name("lastname"));
        lname.clear();

        waitTime();

        // Enter new last name
        lname.sendKeys("User");

        waitTime();

        // Click update/save button
        driver.findElement(
                By.xpath("//button[@class='btn btn-primary' and @name='save']"))
                .click();

        waitTime();

        // Handle update confirmation alert
        Alert updateAlert = driver.switchTo().alert();

        System.out.println("Student Updating Alert Message : "
                + updateAlert.getText());

        waitTime();

        // Accept update alert
        updateAlert.accept();

        waitTime();

        // ==========================
        // LOGOUT PROCESS
        // ==========================

        // Directly access logout page
        driver.get("http://localhost/student/admin/logout.php");

        waitTime();

        // Get current URL after logout
        String currentLogoutURL = driver.getCurrentUrl();

        System.out.println("Current URL after logout: "
                + currentLogoutURL);

        // Validate logout success
        if (currentLogoutURL.contains("index.php")
                || driver.findElements(By.name("username")).size() > 0) {

            System.out.println("Logout Success");

        } else {

            System.out.println("Logout Fail");
        }

        waitTime();

        // Close browser
        driver.quit();
    }
}
