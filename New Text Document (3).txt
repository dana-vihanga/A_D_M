package jar;

import java.time.Duration;

import org.openqa.selenium.Alert;
import org.openqa.selenium.By;
import org.openqa.selenium.JavascriptExecutor;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;

public class test {

    static WebDriver driver;

    public static void waitTime() throws InterruptedException {
        Thread.sleep(5000);
    }

    public static void launchBrowser() {
        System.setProperty("webdriver.chrome.driver",
                "E:\\Downloads\\chromedriver-win64\\chromedriver-win64\\chromedriver.exe");

        driver = new ChromeDriver();
        driver.manage().window().maximize();
        driver.manage().timeouts().pageLoadTimeout(Duration.ofSeconds(10));
    }

    public static void openAdminPage() {
        driver.get("http://localhost/student/admin");
    }

    public static void login(String username, String password)
            throws InterruptedException {

        driver.findElement(By.name("username")).sendKeys(username);
        waitTime();

        driver.findElement(By.name("password")).sendKeys(password);
        waitTime();

        driver.findElement(By.xpath("//button[@type='submit']")).click();

        System.out.println("Login Success");
    }

    public static void navigateToStudents() throws InterruptedException {
        driver.findElement(By.linkText("Students")).click();
        waitTime();
    }

    public static void validateURL() {

        String expectedURL =
                "http://localhost/student/admin/student.php";

        String currentURL = driver.getCurrentUrl();

        if (currentURL.equals(expectedURL)) {
            System.out.println("URL Validation Success");
        } else {
            System.out.println("URL Validation Fail");
        }
    }

    public static void searchStudent(String name)
            throws InterruptedException {

        WebElement search =
                driver.findElement(By.xpath("//input[@type='search']"));

        search.sendKeys(name);
        waitTime();
    }

    public static void addStudent()
            throws InterruptedException {

        driver.findElement(By.id("new_std_btn")).click();
        waitTime();

        driver.findElement(By.name("firstname"))
                .sendKeys("Stanley");

        driver.findElement(By.name("lastname"))
                .sendKeys("Lambert");

        driver.findElement(By.name("grade"))
                .sendKeys("Grade 11");

        driver.findElement(By.name("phone"))
                .sendKeys("0777889090");

        waitTime();

        driver.findElement(By.name("save")).click();

        waitTime();

        Alert alert = driver.switchTo().alert();

        System.out.println(
                "Student Saving Alert Message : "
                        + alert.getText());

        alert.accept();
    }

    public static void deleteStudent()
            throws InterruptedException {

        WebElement deleteBtn = driver.findElement(
                By.xpath("(//button[contains(@class,'remove_student')])[1]"));

        ((JavascriptExecutor) driver)
                .executeScript("arguments[0].click();", deleteBtn);

        waitTime();

        Alert alert = driver.switchTo().alert();

        System.out.println(
                "Student Deleting Alert Message : "
                        + alert.getText());

        alert.dismiss();
    }

    public static void updateStudent()
            throws InterruptedException {

        WebElement editBtn = driver.findElement(
                By.xpath("(//button[contains(@class,'edit_student')])[1]"));

        ((JavascriptExecutor) driver)
                .executeScript("arguments[0].click();", editBtn);

        waitTime();

        WebElement lname =
                driver.findElement(By.name("lastname"));

        lname.clear();
        lname.sendKeys("User");

        waitTime();

        driver.findElement(
                By.xpath("//button[@class='btn btn-primary' and @name='save']"))
                .click();

        waitTime();

        Alert alert = driver.switchTo().alert();

        System.out.println(
                "Student Updating Alert Message : "
                        + alert.getText());

        alert.accept();
    }

    public static void logout()
            throws InterruptedException {

        driver.get("http://localhost/student/admin/logout.php");

        waitTime();

        String currentURL = driver.getCurrentUrl();

        if (currentURL.contains("index.php")
                || driver.findElements(By.name("username")).size() > 0) {

            System.out.println("Logout Success");

        } else {

            System.out.println("Logout Fail");
        }
    }

    public static void closeBrowser() {
        driver.quit();
    }

    public static void main(String[] args)
            throws InterruptedException {

        launchBrowser();

        driver.get("http://localhost/student");

        waitTime();

        openAdminPage();

        login("admin", "admin@123");

        navigateToStudents();

        validateURL();

        searchStudent("Ad");

        addStudent();

        deleteStudent();

        updateStudent();

        logout();

        closeBrowser();
    }
}
