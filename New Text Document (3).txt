    # Prompt user

    jal print_string

    la $a0, even_msg
    jal print_string

end_q2:

.data
.align 2
arry: .word 100, 10, 3, -2, 110, 0, -50, 150, -17, 8
size: .word 10
msg: .asciiz "Smallest number: "

.text
main:
    la $t0, arry       # Array base address
    lw $t1, size       # Load size (10)
    lw $t2, 0($t0)     # $t2 = smallest (Initialize with first element)
    
    li $t3, 1          # Counter i = 1

loop:
    beq $t3, $t1, print_min
    
    # Next element ලිපිනය ගණනය කිරීම (i * 4)
    mul $t4, $t3, 4
    add $t4, $t0, $t4
    lw $t5, 0($t4)     # $t5 = current element
    
    # If current < smallest, update smallest
    blt $t5, $t2, update_min
    
    addi $t3, $t3, 1
    j loop

update_min:
    move $t2, $t5      # Update smallest
    addi $t3, $t3, 1
    j loop

print_min:
    li $v0, 4
    la $a0, msg
    syscall
    
    move $a0, $t2
    li $v0, 1
    syscall
    
    li $v0, 10
    syscall

# Q1_util.asm හි ඇති උපක්‍රම මෙතනට සම්බන්ධ කිරීම
.include "Q1_util.asm"

.text
.globl main

main:
    # 1. Prompt the user for an integer
    la $a0, prompt_msg
    jal print_string      # Q1(b) පාවිච්චි කිරීම
    
    # 2. Get integer from user
    jal prompt_int        # Q1(c) පාවිච්චි කිරීම
    move $t0, $v0         # $t0 = user input
    
    # 3. Check if even or odd (Divide by 2 and check remainder)
    li $t1, 2
    rem $t2, $t0, $t1     # $t2 = $t0 % 2
    
    # ඉතිරිය 0 නම් number එක Even, නැත්නම් Odd
    beq $t2, $zero, show_even
    
    # If odd:
    la $a0, odd_msg
    jal print_string
    j end_program

show_even:
    la $a0, even_msg
    jal print_string

end_program:
    # 4. Exit
    j exit                # Q1(a) වෙත යාම

.data
prompt_num1: .asciiz "Enter Number 1: "
prompt_num2: .asciiz "Enter Number 2: "

menu_title:  .asciiz "\nMenu\n"
menu_add:    .asciiz "1 - Addition\n"
menu_sub:    .asciiz "2 - Subtraction\n"
menu_mul:    .asciiz "3 - Multiplication\n"
menu_div:    .asciiz "4 - Division (Quotient)\n"
menu_mod:    .asciiz "5 - Modulus (Remainder)\n"
prompt_op:   .asciiz "Select the operation from the Menu: "

sign_add:    .asciiz " + "
sign_sub:    .asciiz " - "
sign_mul:    .asciiz " x "
sign_div:    .asciiz " / "
sign_mod:    .asciiz " % "
sign_equal:  .asciiz " = "

div_zero_err: .asciiz "Error: Division by zero is not allowed.\n"
invalid_err:  .asciiz "Error: Invalid Menu Selection.\n"

.text
.globl main

main:
    # 1. Enter Number 1
    li $v0, 4
    la $a0, prompt_num1
    syscall
    li $v0, 5
    syscall
    move $t0, $v0        # $t0 = Number 1

    # 2. Enter Number 2
    li $v0, 4
    la $a0, prompt_num2
    syscall
    li $v0, 5
    syscall
    move $t1, $v0        # $t1 = Number 2

    # 3. Display Menu
    li $v0, 4
    la $a0, menu_title
    syscall
    la $a0, menu_add
    syscall
    la $a0, menu_sub
    syscall
    la $a0, menu_mul
    syscall
    la $a0, menu_div
    syscall
    la $a0, menu_mod
    syscall

    # 4. Prompt for operation selection
    la $a0, prompt_op
    syscall
    li $v0, 5
    syscall
    move $t2, $v0        # $t2 = Choice (1, 2, 3, 4, 5)

    # 5. Jump based on Menu Selection
    li $t3, 1
    beq $t2, $t3, do_add
    li $t3, 2
    beq $t2, $t3, do_sub
    li $t3, 3
    beq $t2, $t3, do_mul
    li $t3, 4
    beq $t2, $t3, do_div
    li $t3, 5
    beq $t2, $t3, do_mod

    # If none of the above, show invalid choice error
    li $v0, 4
    la $a0, invalid_err
    syscall
    j exit_program

do_add:
    add $t4, $t0, $t1    # $t4 = result
    la $s0, sign_add     # Store operator sign address in $s0
    j print_result

do_sub:
    sub $t4, $t0, $t1
    la $s0, sign_sub
    j print_result

do_mul:
    mul $t4, $t0, $t1
    la $s0, sign_mul
    j print_result

do_div:
    # Check for division by zero
    beq $t1, $zero, error_div_zero
    div $t4, $t0, $t1
    la $s0, sign_div
    j print_result

do_mod:
    # Check for modulus by zero
    beq $t1, $zero, error_div_zero
    rem $t4, $t0, $t1
    la $s0, sign_mod
    j print_result

error_div_zero:
    li $v0, 4
    la $a0, div_zero_err
    syscall
    j exit_program

print_result:
    # Output Format: [Num1] [Operator] [Num2] = [Result]
    
    # Print Num1
    li $v0, 1
    move $a0, $t0
    syscall

    # Print Operator Sign
    li $v0, 4
    move $a0, $s0
    syscall

    # Print Num2
    li $v0, 1
    move $a0, $t1
    syscall

    # Print "=" Sign
    li $v0, 4
    la $a0, sign_equal
    syscall

    # Print Result
    li $v0, 1
    move $a0, $t4
    syscall

exit_program:
    # Exit system call
    li $v0, 10
    syscall

.data
prompt_n: .asciiz "Enter n: "
result_n: .asciiz "Factorial is: "

.text
.globl main
main:
    # Input 'n' from keyboard
    li $v0, 4
    la $a0, prompt_n
    syscall
    li $v0, 5
    syscall
    move $a0, $v0        # $a0 ඇතුළට 'n' අගය දානවා (Argument එක විදිහට)

    jal factorial        # Factorial function එක call කිරීම
    move $t0, $v0        # පිළිතුර $t0 වෙත තබා ගැනීම

    # පිළිතුර ප්‍රින්ට් කිරීම
    li $v0, 4
    la $a0, result_n
    syscall
    
    move $a0, $t0
    li $v0, 1
    syscall

    # Exit
    li $v0, 10
    syscall

# ---- Recursive Factorial Function ----
factorial:
    subu $sp, $sp, 8     # Stack එකේ ඉඩ වෙන් කිරීම
    sw $ra, 4($sp)       # Return Address එක සේව් කිරීම
    sw $a0, 0($sp)       # වත්මන් 'n' අගය සේව් කිරීම

    # Base Case: n <= 0 නම් 1ක් රිටන් කරන්න
    blez $a0, base_case

    # Recursive Step: fact(n-1)
    sub $a0, $a0, 1
    jal factorial
    
    # කෝඩ් එක ආපසු හැරෙන විට (Unwinding stack)
    lw $a0, 0($sp)       # පරණ 'n' අගය නැවත ලෝඩ් කිරීම
    lw $ra, 4($sp)       # Return Address එක නැවත ලෝඩ් කිරීම
    addi $sp, $sp, 8     # Stack එක යථා තත්ත්වයට පත් කිරීම

    mul $v0, $a0, $v0    # n * fact(n-1)
    jr $ra

base_case:
    li $v0, 1            # 0! හෝ 1! වලට පිළිතුර 1යි
    lw $ra, 4($sp)
    addi $sp, $sp, 8
    jr $ra
